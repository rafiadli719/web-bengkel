// ===============================================
// KALKULATOR TARIF JEMPUT ANTAR DENGAN AUTO DISTANCE
// ===============================================

// Koordinat Cabang (dari PHP)
const latCabang = <?php echo !empty($lat_cabang) ? $lat_cabang : 'null'; ?>;
const longCabang = <?php echo !empty($long_cabang) ? $long_cabang : 'null'; ?>;

// Koordinat Pelanggan (akan diupdate saat pilih kendaraan)
let latPelanggan = null;
let longPelanggan = null;

// Auto-calculate jarak saat halaman load jika koordinat sudah ada
document.addEventListener('DOMContentLoaded', function() {
    // Ambil koordinat pelanggan dari data yang sudah ada (jika ada)
    <?php if (!empty($customer_data['klat']) && !empty($customer_data['klong'])): ?>
    latPelanggan = <?php echo $customer_data['klat']; ?>;
    longPelanggan = <?php echo $customer_data['klong']; ?>;

    if (latCabang && longCabang && latPelanggan && longPelanggan) {
        autoCalculateDistance();
    }
    <?php endif; ?>
});

// Function auto hitung jarak
function autoCalculateDistance() {
    if (!latCabang || !longCabang) {
        console.log('⚠️ Koordinat cabang belum tersedia! Update data cabang terlebih dahulu.');
        return;
    }

    if (!latPelanggan || !longPelanggan) {
        console.log('⚠️ Koordinat pelanggan belum tersedia!');
        return;
    }

    // Hitung jarak menggunakan Haversine formula
    const jarak = calculateDistance(latCabang, longCabang, latPelanggan, longPelanggan);

    // Set nilai jarak ke input field (round to 1 decimal)
    document.getElementById('txtjarak').value = jarak.toFixed(1);

    // Auto hitung tarif
    hitungTarif();

    // Tampilkan notifikasi
    const jarakText = jarak.toFixed(1);
    console.log(`✅ Jarak otomatis dihitung: ${jarakText} KM (dari <?php echo $nama_cabang; ?> ke lokasi pelanggan)`);

    // Tampilkan info di form
    const infoDiv = document.getElementById('hasilTarif');
    if (infoDiv) {
        const existingInfo = infoDiv.querySelector('.info-jarak-otomatis');
        if (existingInfo) existingInfo.remove();

        const info = document.createElement('div');
        info.className = 'info-jarak-otomatis';
        info.style.cssText = 'background: #d4edda; padding: 10px; border-radius: 5px; margin-top: 10px; border: 1px solid #c3e6cb;';
        info.innerHTML = `<i class="fa fa-check-circle" style="color: #28a745;"></i> <strong>Jarak otomatis dihitung:</strong> ${jarakText} KM<br>
                         <small>Dari ${latCabang},${longCabang} (<?php echo $nama_cabang; ?>) ke ${latPelanggan},${longPelanggan} (Pelanggan)</small>`;
        infoDiv.appendChild(info);
    }
}

function hitungTarif() {
    const jarak = parseFloat(document.getElementById('txtjarak').value) || 0;

    let tarifJalan = 0;
    let tarifMogok = 0;

    if (jarak > 1.0) {
        // Perhitungan Motor Jalan
        tarifJalan = hitungTarifJemput(jarak, 'jalan');

        // Perhitungan Motor Mogok
        tarifMogok = hitungTarifJemput(jarak, 'mogok');
    }

    // Update display
    document.getElementById('tarifJalan').textContent = formatRupiah(tarifJalan);
    document.getElementById('tarifMogok').textContent = formatRupiah(tarifMogok);
    document.getElementById('hasilTarif').style.display = 'block';

    // Set hidden field berdasarkan kondisi motor yang dipilih
    const kondisi = document.querySelector('input[name="txtkondisi"]:checked').value;
    const tarifTerpilih = kondisi === 'jalan' ? tarifJalan : tarifMogok;
    document.getElementById('txttarif').value = tarifTerpilih;
}

// Event listener untuk tombol hitung
document.getElementById('btnHitungTarif').addEventListener('click', function() {
    // Jika jarak masih 0, coba hitung otomatis
    const jarak = parseFloat(document.getElementById('txtjarak').value) || 0;
    if (jarak === 0 && latCabang && longCabang && latPelanggan && longPelanggan) {
        autoCalculateDistance();
    } else {
        hitungTarif();
    }
});

// Auto-hitung saat jarak berubah manual
document.getElementById('txtjarak').addEventListener('input', function() {
    if (this.value > 0) {
        hitungTarif();
    }
});

// Auto-update tarif saat kondisi motor berubah
document.querySelectorAll('input[name="txtkondisi"]').forEach(function(radio) {
    radio.addEventListener('change', function() {
        const jarak = parseFloat(document.getElementById('txtjarak').value) || 0;
        if (jarak > 0) {
            hitungTarif();
        }
    });
});

// Update koordinat pelanggan saat Google Maps link diupdate
